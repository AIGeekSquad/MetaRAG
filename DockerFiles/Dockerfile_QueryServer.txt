FROM python:3.11

WORKDIR /app

# Install system dependencies
RUN apt-get update && apt-get install -y \
    gcc \
    python3-dev \
    musl-dev \
    build-essential \
    pipx

# Install uv using pipx
RUN pipx install uv
ENV PATH="/root/.local/bin:$PATH"

# Copy project files and install dependencies
COPY pyproject.toml .
RUN uv sync
RUN uv pip install --upgrade pymupdf

# Copy application code
COPY . .

EXPOSE 8000
CMD ["uvicorn", "query_server_process:app", "--host", "0.0.0.0", "--port", "8000"]
